HEADERS = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36',
    'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
    'Accept-Language': 'id-ID,id;q=0.9,en-US;q=0.8,en;q=0.7',
}

# --- FUNGSI AMBIL RSS ---
def dapatkan_feed_rss(url_rss):
    try:
        response = requests.get(url_rss, headers=HEADERS, timeout=12, verify=False)
        if response.status_code == 200:
            return feedparser.parse(response.content)
    except Exception:
        pass
    return feedparser.parse(url_rss)

# --- FUNGSI EKSTRAKSI URL ASLI (JIKA VIA GOOGLE NEWS) ---
def dapatkan_url_asli(url_target):
    if "news.google.com" in url_target:
        try:
            resp = requests.get(url_target, headers=HEADERS, timeout=10, verify=False, allow_redirects=True)
            return resp.url
        except Exception:
            return url_target
    return url_target

# --- FUNGSI SCRAPING KONTEN UNIVERSAL ---
def ambil_isi_berita(url_input, tag_html, class_html, butuh_page_all):
    try:
        url_asli = dapatkan_url_asli(url_input)
        link_target = url_asli + "?page=all" if butuh_page_all else url_asli
        
        response = requests.get(link_target, headers=HEADERS, timeout=12, verify=False, allow_redirects=True)
        
        if response.status_code == 200:
            soup = BeautifulSoup(response.text, 'html.parser')
            
            # 1. Cari berdasarkan tag & class bawaan
            artikel_body = soup.find(tag_html, class_=class_html)
            
            # 2. Fallback Spesifik Domain
            if not artikel_body:
                url_check = response.url.lower()
                if "idnfinancials.com" in url_check:
                    artikel_body = (
                        soup.find('article', class_='article-content') or
                        soup.find('div', class_='article-body') or
                        soup.find('div', class_='content-detail') or
                        soup.find('article')
                    )
                elif "cnbcindonesia.com" in url_check:
                    artikel_body = (
                        soup.find('div', class_='detail_text') or
                        soup.find('div', class_='detail-text') or
                        soup.find('article')
                    )
                elif "investor.id" in url_check:
                    artikel_body = (
                        soup.find('div', class_='body-content') or
                        soup.find('div', class_='content-detail') or
                        soup.find('div', class_='article-body') or
                        soup.find('div', class_='col-12 col-lg-8') or
                        soup.find('article')
                    )

            # 3. Ambil Paragraf dari Elemen Pembungkus
            if artikel_body:
                paragraf = artikel_body.find_all('p')
                teks_paragraf = [p.text.strip() for p in paragraf if len(p.text.strip()) > 20]
                if teks_paragraf:
                    return "\n\n".join(teks_paragraf)
            
            # 4. Fallback Universal (Semua paragraf panjang di web)
            semua_p = soup.find_all('p')
            teks_universal = []
            for p in semua_p:
                txt = p.text.strip()
                if len(txt) > 30 and not re.search(r'(cookie|privacy|rights reserved|baca juga|baca selengkapnya)', txt, re.IGNORECASE):
                    teks_universal.append(txt)
            
            if teks_universal:
                return "\n\n".join(teks_universal)
                    
            return "Konten tidak dapat diekstrak (Struktur HTML dilindungi/berubah)."
        return f"Gagal akses. Status: {response.status_code}"
    except Exception as e:
        return f"Error: {e}"

# ==========================================
st.set_page_config(page_title="Radar Investasi Multi", layout="wide")

st.sidebar.title("⚙️ Pengaturan Radar")
st.sidebar.write("Pilih sumber berita yang ingin dipindai:")

portal_terpilih = st.sidebar.multiselect(
    "Pilih Portal/Kanal Berita (Bisa lebih dari 1):",
    options=list(aturan_portal.keys()),
    default=[
        "IDNFinancials",
        "CNBC Indonesia (Market)", 
        "CNBC Indonesia (News)",
        "Investor.id (Market)",
        "Kontan Investasi"
    ]
)

target_jumlah = st.sidebar.number_input(
    "Maksimal berita per portal:", 
    min_value=1, max_value=50, value=10
)

st.sidebar.markdown("---")
st.sidebar.info("Aplikasi lokal berjalan tanpa batasan timeout server awan.")

st.title("📈 Radar Berita Portofolio (Mode Multi-Portal)")
st.caption("Target Pemantauan Emiten, Aset, Regulasi & Sentimen Pasar:")
st.code(", ".join([k.upper() for k in kata_kunci_portofolio]))

if st.button("Mulai Radar Multi-Portal!", type="primary"):
    
    if len(portal_terpilih) == 0:
        st.warning("⚠️ Silakan pilih minimal satu portal/kanal berita di sidebar terlebih dahulu.")
    else:
        kumpulan_data_global = []
        status_teks = st.empty()
        progress_bar = st.progress(0)
        total_portal = len(portal_terpilih)
        
        for idx, nama_portal in enumerate(portal_terpilih):
            status_teks.info(f"🔄 Sedang memindai: **{nama_portal}** (Kanal ke-{idx+1} dari {total_portal})")
            
            aturan = aturan_portal[nama_portal]
            feed = dapatkan_feed_rss(aturan["rss"])
            
            if feed and hasattr(feed, 'entries') and len(feed.entries) > 0:
                jumlah_didapat = 0 
                for entry in feed.entries:
                    judul = entry.get('title', 'N/A')
                    link = entry.get('link', 'N/A')
                    tanggal = entry.get('published', 'N/A')
                    
                    judul_lower = judul.lower()
                    cocok = any(kunci in judul_lower for kunci in kata_kunci_portofolio)
                    
                    if not cocok:
                        continue 
                    
                    isi = ambil_isi_berita(link, aturan["tag"], aturan["class"], aturan["butuh_page_all"])
                    
                    # SEMUA KOLOM DILENGKAPI UNTUK TAMPILAN WEB LAYAR
                    kumpulan_data_global.append({
                        "Sumber": nama_portal,
                        "Trigger/Emiten": next((k.upper() for k in kata_kunci_portofolio if k in judul_lower), "UMUM"),
                        "Judul": judul,
                        "Tanggal": tanggal,
                        "Link": link,
                        "Isi Berita": isi
                    })
                    
                    jumlah_didapat += 1
                    time.sleep(1) 
                    
                    if jumlah_didapat >= target_jumlah:
                        break
            else:
                st.error(f"Gagal membaca RSS dari {nama_portal}.")
                
            progress_bar.progress((idx + 1) / total_portal)
        
        status_teks.empty()
        
        if len(kumpulan_data_global) > 0:
            st.success(f"Radar selesai! Menemukan total {len(kumpulan_data_global)} berita relevan dari {total_portal} portal/kanal.")
            df = pd.DataFrame(kumpulan_data_global)
            
            # 1. Menampilkan TABEL LENGKAP di layar Web Streamlit
            st.dataframe(df, use_container_width=True)
            
            # 2. FILTER KHUSUS CSV: Hanya menyaring Judul, Tanggal, dan Isi Berita saja untuk file download
            df_csv = df[['Judul', 'Tanggal', 'Isi Berita']]
            csv_data = df_csv.to_csv(index=False).encode('utf-8')
            
            st.download_button(
                label="📥 Unduh CSV Ringkas (Hanya Judul, Tanggal & Isi)",
                data=csv_data,
                file_name='laporan_berita_ringkas.csv',
                mime='text/csv',
            )
        else:
            st.warning("Belum ada berita yang memuat kata kuncimu di kanal-kanal tersebut saat ini.")
